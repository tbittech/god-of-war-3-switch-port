# Envoltorio: compila el runtime de ps3recomp para Switch con la capa nx/compat.
cmake_minimum_required(VERSION 3.20)
project(gow3_rt_nx LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(NX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(PS3RECOMP_DIR "${NX_DIR}/../ref/ps3recomp" ABSOLUTE)
include_directories(SYSTEM "${NX_DIR}/compat/include")
include_directories("${NX_DIR}/compat")
include_directories(SYSTEM "$ENV{DEVKITPRO}/portlibs/switch/include")
add_compile_options($<$<COMPILE_LANGUAGE:C>:-Wno-error=incompatible-pointer-types>)
add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:SHELL:-include nx_compat.h>")
add_subdirectory("${PS3RECOMP_DIR}" "${CMAKE_BINARY_DIR}/ps3recomp_sdk")
add_library(gow3_nx_compat STATIC "${NX_DIR}/compat/nx_compat.cpp" "${NX_DIR}/compat/nx_ucontext.S" "${NX_DIR}/gm/guest_memory_switch.cpp")
